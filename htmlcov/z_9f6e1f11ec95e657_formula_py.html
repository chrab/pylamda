1"""Formula mass of a molecule name as written in LAMDA files (for check P04).""" 

2 

3from __future__ import annotations 

4 

5import re 

6 

7#: Standard atomic weight [Da] of the elements H to Zn: the abridged values of IUPAC 2021 

8#: (Prohaska et al. 2022, Pure Appl. Chem. 94, 573, Table 2; for elements with an 

9#: interval, the conventional value). Copied from astrolinedata (``species.py``). 

10#: 

11#: :meta hide-value: 

12STANDARD_ATOMIC_WEIGHT: dict[str, float] = { 

13 "H": 1.008, "He": 4.0026, "Li": 6.94, "Be": 9.0122, "B": 10.81, "C": 12.011, 

14 "N": 14.007, "O": 15.999, "F": 18.998, "Ne": 20.180, "Na": 22.990, "Mg": 24.305, 

15 "Al": 26.982, "Si": 28.085, "P": 30.974, "S": 32.06, "Cl": 35.45, "Ar": 39.95, 

16 "K": 39.098, "Ca": 40.078, "Sc": 44.956, "Ti": 47.867, "V": 50.942, "Cr": 51.996, 

17 "Mn": 54.938, "Fe": 55.845, "Co": 58.933, "Ni": 58.693, "Cu": 63.546, "Zn": 65.38, 

18} # fmt: skip 

19 

20# deuterium is written as its own symbol (2H, 2.014 Da) 

21_EXTRA = {"D": 2.014} 

22# spin isomers and structural prefixes: o-H2O, p-NH3, A-CH3OH, E-CH3OH, c-C3H2, l-C3H 

23_PREFIX = re.compile(r"^(?:ortho|para|meta|[opmAEcl])-") 

24_ATOM = re.compile(r"(\d+)?([A-Z][a-z]?)(\d*)") 

25 

26 

27def formula_mass(name: str) -> float | None: 

28 """Mass [Da] of a molecule from its name (``"CO"``, ``"13CO"``, ``"C18O"``, ``"o-H3O+"``). 

29 

30 The name is read as a chemical formula, after an optional prefix (``o-``, ``p-``, 

31 ``A-``, ``E-``, ``c-``, ``l-``, ...) and before an optional charge (``+``, ``++``, 

32 ``-``). A number of 10 or more before an element is a mass number (``13C``, ``18O``, 

33 also in ``C18O`` and ``H13CN``); a number below 10 after an element is a count 

34 (``H2``, ``HC3N``). An isotope counts with its mass number, the other elements with 

35 their standard atomic weight. The charge is ignored (the electron mass is below the 

36 precision of the check). 

37 

38 Parameters 

39 ---------- 

40 name : str 

41 The molecule name, e.g. the first word of line 2 of a LAMDA file. 

42 

43 Returns 

44 ------- 

45 float or None 

46 The mass [Da], or None if the name does not read as a formula of known elements. 

47 """ 

48 text = _PREFIX.sub("", name.strip()) 

49 text = re.sub(r"[+-]+$", "", text) 

50 if not text: 

51 return None 

52 mass = 0.0 

53 pending: int | None = None # a mass number written after the previous element (C18O) 

54 pos = 0 

55 while pos < len(text): 

56 m = _ATOM.match(text, pos) 

57 if not m: 

58 return None 

59 iso_text, element, count_text = m.groups() 

60 weight = STANDARD_ATOMIC_WEIGHT.get(element, _EXTRA.get(element)) 

61 if weight is None: 

62 return None 

63 if iso_text and pending is not None: 63 ↛ 64line 63 didn't jump to line 64 because the condition on line 63 was never true

64 return None 

65 iso = int(iso_text) if iso_text else pending 

66 pending = None 

67 count = int(count_text) if count_text else 1 

68 if count >= 10 and m.end() < len(text): # "C18O": 18 is the mass number of the next atom 

69 pending, count = count, 1 

70 if count < 1: 

71 return None 

72 if iso is not None and not (0.5 * weight <= iso <= 2 * weight + 2): 72 ↛ 73line 72 didn't jump to line 73 because the condition on line 72 was never true

73 return None 

74 mass += count * (float(iso) if iso is not None else weight) 

75 pos = m.end() 

76 return None if pending is not None else mass
